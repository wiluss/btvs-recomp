cmake_minimum_required(VERSION 3.20)
project(buffy_macos C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(XBOXRECOMP_ENHANCE "Build the pinned graphics settings and stats layer" ON)
add_executable(buffy-host-probe src/host/probe.c)
target_compile_options(buffy-host-probe PRIVATE -Wall -Wextra -Werror)

# Generic dependencies and backend come exclusively from the pinned submodule.
if(NOT WIN32)
    find_package(OpenSSL REQUIRED)
endif()
add_subdirectory(vendor/xboxrecomp)

option(BUFFY_BUILD_GAME "Build experimental executable from locally generated C" OFF)
if(BUFFY_BUILD_GAME)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/prepare_host.py"
        RESULT_VARIABLE HOST_RESULT)
    if(NOT HOST_RESULT EQUAL 0)
        message(FATAL_ERROR "Host template preparation failed")
    endif()
    file(GLOB GAME_C CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/generated/recomp/gen/*.c")
    if(NOT GAME_C)
        message(FATAL_ERROR "Run make generate before building the game")
    endif()
    add_executable(buffy-macos src/host/controls.c src/host/widescreen.c src/host/arenas.c generated/host/main.c generated/host/recomp_manual.c ${GAME_C})
    target_include_directories(buffy-macos PRIVATE src/host generated/host generated/recomp/gen
        vendor/xboxrecomp/include vendor/xboxrecomp/src)
    target_link_libraries(buffy-macos PRIVATE xboxrecomp)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        set_property(SOURCE ${GAME_C} APPEND PROPERTY COMPILE_OPTIONS -Wno-parentheses-equality)
    endif()
endif()
